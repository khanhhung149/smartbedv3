cmake_minimum_required(VERSION 3.16)
project(smartbed_stream_processor_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(smartbed_stream_processor src/smartbed_stream_processor.cpp)
if (MSVC)
  target_compile_options(smartbed_stream_processor PRIVATE /W4)
else()
  target_compile_options(smartbed_stream_processor PRIVATE -Wall -Wextra -pedantic)
endif()
